# 07 · 组合根与启动

> 组合根（composition root）的职责、DI 容器装配、lifespan 启动钩子、ensure 脚本、可选 gateway。

> **基于 docs/python-refactor/07-组合根与启动.md 模板**。
> 与原模板差异：DI 容器支持按 tenant / workspace 注入 scope；ensure 阶段创建通用 Agent 模板；多租户上下文在容器启动时绑定。

---

## 7.1 组合根定位

组合根是**应用启动的唯一入口**，负责：

1. 解析 Settings（`pydantic-settings`）
2. 构建 DI 容器（按模块注册服务）
3. 装配 lifespan 钩子（启动 / 关闭）
4. 注册中间件链（顺序敏感）
5. 挂载各模块的 FastAPI 路由
6. 触发 ensure 脚本（建通用资源）
7. 启动事件总线 / 远程 runtime 客户端

```
composition/qzdap_app/src/qzdap/composition/
├── __init__.py
├── main.py                # create_app() 工厂
├── lifespan.py            # lifespan context manager
├── container.py           # DI 容器装配
├── settings.py            # Settings（pydantic-settings）
├── ensure.py              # 启动 ensure
├── middleware.py          # 默认中间件聚合
└── route_loader.py        # 模块路由自动挂载
```

## 7.3 DI 容器

### 7.3.1 选用

采用自研极简容器（< 100 行），避免引入 `dependency-injector` 等重框架。

```python
# libs/kernel/src/qzdap_kernel/container.py
from typing import Type, Callable, Any


class Container:
    def __init__(self):
        self._factories: dict[type, Callable[["Container"], Any]] = {}
        self._singletons: dict[type, Any] = {}

    def register(self, iface: type, factory: Callable[["Container"], Any]) -> None:
        self._factories[iface] = factory

    def register_singleton(self, iface: type, instance: Any) -> None:
        self._singletons[iface] = instance

    def resolve(self, iface: type) -> Any:
        if iface in self._singletons:
            return self._singletons[iface]
        if iface not in self._factories:
            raise KeyError(f"No factory for {iface}")
        instance = self._factories[iface](self)
        return instance

    def override(self, iface: type, factory: Callable[["Container"], Any]) -> None:
        """测试时覆盖工厂"""
        self._factories[iface] = factory
```

### 7.3.2 容器装配

```python
# composition/qzdap_app/src/qzdap/composition/container.py
from qzdap_kernel import Container
from libs.auth import JWTService, APIKeyRepo
from libs.persistence import SessionFactory, UnitOfWork
from libs.messaging import EventBus, InProcessBus, RedisStreamBus
from libs.llm import LLMRegistry, LLMRouter
from libs.sandbox import Sandbox, LocalSandbox, DockerSandbox
from libs.vector import VectorStore, PgVectorStore
from libs.observability import setup_logging, setup_tracing

# 各模块
from modules.identity.application.services import IdentityService
from modules.agent_runtime.application.services import AgentRuntimeService
from modules.tool.application.services import ToolService
from modules.skill.application.services import SkillService
from modules.memory.application.services import MemoryService
from modules.governance.application.services import GovernanceService
from modules.governance.application.ports import PolicyPort
# ...

def build_container() -> Container:
    c = Container()

    # ----- 基础设施 -----
    c.register_singleton(Settings, Settings())

    sf = SessionFactory(settings.database_url)
    c.register_singleton(SessionFactory, sf)

    redis = Redis.from_url(settings.redis_url, max_connections=50)
    c.register_singleton(Redis, redis)

    # 事件总线
    if settings.runtime_mode == "local" or settings.is_demo:
        c.register_singleton(EventBus, InProcessBus(queue_size=1000))
    else:
        c.register_singleton(EventBus, RedisStreamBus(redis, stream_key="qzdap:events"))

    # LLM
    c.register_singleton(LLMRegistry, LLMRegistry.from_settings(settings))
    c.register_singleton(LLMRouter, LLMRouter(c.resolve(LLMRegistry), default=settings.default_model))

    # 沙箱
    if settings.runtime_mode == "local":
        c.register_singleton(Sandbox, LocalSandbox())
    else:
        c.register_singleton(Sandbox, DockerSandbox(settings.skill_runtime_url))

    # 向量存储
    c.register_singleton(VectorStore, PgVectorStore(sf))

    # ----- 横切模块 -----
    c.register(IdentityService, lambda c: IdentityService(
        uow_factory=lambda: UnitOfWork(c.resolve(SessionFactory)),
        ...
    ))
    c.register(GovernanceService, lambda c: GovernanceService(
        policy_repo=...,
        approval_repo=...,
        quota_repo=...,
        events=c.resolve(EventBus),
    ))
    c.register(PolicyPort, lambda c: GovernanceAdapter(c.resolve(GovernanceService)))

    # ----- 起步四模块 -----
    c.register(AgentRuntimeService, lambda c: AgentRuntimeService(
        session_repo=...,
        turn_repo=...,
        llm=c.resolve(LLMRouter),
        tool=ToolPortAdapter(c.resolve(ToolService)),
        skill=SkillPortAdapter(c.resolve(SkillService)),
        memory=MemoryPortAdapter(c.resolve(MemoryService)),
        knowledge=KnowledgePortAdapter(...),
        policy=c.resolve(PolicyPort),
        events=c.resolve(EventBus),
    ))

    c.register(ToolService, lambda c: ToolService(
        tool_repo=...,
        invocation_repo=...,
        mcp_runtime=...,
        policy=c.resolve(PolicyPort),
        events=c.resolve(EventBus),
    ))

    c.register(SkillService, lambda c: SkillService(
        package_repo=...,
        install_repo=...,
        invocation_repo=...,
        sandbox=c.resolve(Sandbox),
        run_token_issuer=...,
        policy=c.resolve(PolicyPort),
        events=c.resolve(EventBus),
    ))

    c.register(MemoryService, lambda c: MemoryService(
        entry_repo=...,
        embeddings=EmbeddingAdapter(c.resolve(VectorStore)),
        vector=c.resolve(VectorStore),
        policy=c.resolve(PolicyPort),
        events=c.resolve(EventBus),
    ))

    return c
```

### 7.3.3 多租户 scope（按需）

```python
class TenantScope:
    """请求级别的 tenant 容器：在请求内继承主容器，但 query 已绑定 tenant"""

    def __init__(self, parent: Container, tenant_id: TenantId, workspace_id: WorkspaceId):
        self._parent = parent
        self._tenant_id = tenant_id
        self._workspace_id = workspace_id

    def resolve(self, iface: type) -> Any:
        # ORM 仓储在创建时已绑定 tenant_id 过滤（通过 contextvars）
        return self._parent.resolve(iface)
```

> ORM 自动注入 tenant_id 过滤（见 [03-共享内核 第 3.5.2 节](03-共享内核.md)），
> 无需每个仓储手动加参数。

## 7.4 lifespan

```python
# composition/qzdap_app/src/qzdap/composition/lifespan.py
from contextlib import asynccontextmanager


@asynccontextmanager
async def lifespan(container: Container):
    settings = container.resolve(Settings)

    # 1. 日志 / 追踪
    setup_logging(env=settings.env, level=settings.log_level)
    setup_tracing("qzdap-app", otlp_endpoint=settings.otlp_endpoint)

    log.info("qzdap.app.starting", env=settings.env, instance_id=settings.instance_id)

    # 2. 启动 ensure（建通用资源）
    if settings.is_production or settings.env in ("development", "staging"):
        await ensure_default_resources(container)

    # 3. 启动事件总线
    bus = container.resolve(EventBus)
    await bus.start()

    # 4. 启动事件订阅者（governance / observability_module）
    install_event_subscribers(container, bus)

    # 5. 启动远程 runtime 客户端（如启用）
    if settings.runtime_mode == "remote":
        await warmup_runtimes(container)

    # 6. 健康检查就绪
    container.state["ensure_done"] = True

    log.info("qzdap.app.ready", port=settings.app_port)

    try:
        yield
    finally:
        log.info("qzdap.app.stopping")
        await bus.stop()
        await container.resolve(SessionFactory).close()
        await container.resolve(Redis).aclose()
```

## 7.5 中间件链

```python
# composition/qzdap_app/src/qzdap/composition/middleware.py
from starlette.middleware import Middleware
from libs.http import (
    ErrorEnvelopeMiddleware,
    RequestLogMiddleware,
    TraceIdMiddleware,
    MetricsMiddleware,
    RateLimitMiddleware,
    CORSMiddleware,
    AuthMiddleware,
    TenantGuardMiddleware,
    WorkspaceGuardMiddleware,
)


def default_middlewares(container: Container) -> list[Middleware]:
    return [
        Middleware(ErrorEnvelopeMiddleware),
        Middleware(RequestLogMiddleware),
        Middleware(TraceIdMiddleware),
        Middleware(MetricsMiddleware),
        Middleware(RateLimitMiddleware, container=container),
        Middleware(CORSMiddleware),
        Middleware(AuthMiddleware, container=container),  # JWT 或 API Key
        Middleware(TenantGuardMiddleware),  # X-Tenant-Id 一致性
        Middleway(WorkspaceGuardMiddleware),  # X-Workspace-Id ∈ Principal
    ]
```

**顺序原则**：
1. `ErrorEnvelopeMiddleware` 最外层（捕获所有异常）
2. `RequestLogMiddleware` 早绑定 trace_id
3. `AuthMiddleware` 解析 Principal
4. `TenantGuard` / `WorkspaceGuard` 校验多租户
5. `MetricsMiddleware` 完整记录耗时
6. `RateLimitMiddleware` 按 tenant 限流

## 7.6 ensure 脚本

```python
# composition/qzdap_app/src/qzdap/composition/ensure.py
async def ensure_default_resources(container: Container) -> None:
    settings = container.resolve(Settings)

    # 1. 通用 Agent 模板
    if settings.ensure_default_agent:
        await ensure_default_agent(container)

    # 2. 通用 Skill 包（可选）
    if settings.ensure_default_skill:
        await ensure_default_skill(container)

    # 3. 平台级角色 / 权限
    await ensure_platform_roles(container)

    # 4. demo 环境的演示租户
    if settings.is_demo:
        await ensure_demo_tenant(container)


async def ensure_default_agent(container: Container) -> None:
    agent_factory = container.resolve(AgentFactoryService)
    if not await agent_factory.has_template("general_assistant"):
        template = AgentTemplate(
            id="tpl_general",
            name="general_assistant",
            description="通用企业助手",
            system_prompt="你是一名通用企业助手...",
            default_model="gpt-4o",
            tools=["search_web", "fetch_url"],
            skills=[],
            metadata={},
        )
        await agent_factory.publish(template, version="1.0.0")
```

## 7.8 create_app 工厂

```python
# composition/qzdap_app/src/qzdap/composition/main.py
from fastapi import FastAPI


def create_app() -> FastAPI:
    settings = Settings()
    container = build_container(settings)

    app = FastAPI(
        title="Enterprise-Agent-OS",
        version="0.1.0",
        description="Enterprise AI Agent Platform",
        lifespan=lifespan(container),
        docs_url="/docs",
        redoc_url="/redoc",
    )

    # 中间件（顺序敏感）
    for m in default_middlewares(container):
        app.add_middleware(m)

    # 健康检查 + 指标（无 prefix，无鉴权）
    mount_health_routers(app, container)

    # 模块路由
    mount_module_routers(app, container)

    return app


# 入口
app = create_app()
```

## 7.9 可选网关

`gateways/qzdap_gateway/` 与原模板结构一致，提供：

- 路由转发（`/` → `app_upstream`）
- 限流（Redis token bucket）
- 鉴权（可在网关层校验 JWT 减少下游负担）
- 跨域
- 指标聚合

启动方式：

```bash
# 单体（无网关）
uv run uvicorn qzdap.composition.main:create_app --port 8100

# 经网关
uv run uvicorn qzdap.composition.main:create_app --port 8100 &
uv run uvicorn qzdap.gateway.main:create_gateway --port 8089
```

或 docker-compose `--profile with-gateway`。

## 7.11 反模式

| 反例 | 修正 |
|---|---|
| `from modules.collab.application.services import X` 在 main 中 | 通过 `Container.resolve(X)` |
| 各模块独立注册中间件 | 在组合根统一注册，避免顺序错乱 |
| 在路由函数中 `import` 服务 | 通过 `Depends(get_service)` |
| ensure 脚本同步阻塞 | 用 async + 后台任务 |
| 远程 runtime 未 warmup 就接流量 | lifespan 中预热 |
