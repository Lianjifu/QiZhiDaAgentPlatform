## 3.2 `libs/kernel` — 上下文与身份

### 3.2.1 `Principal`

```python
# libs/kernel/src/qzdap_kernel/principal.py
from dataclasses import dataclass
from typing import Iterable


@dataclass(frozen=True)
class Principal:
    user_id: str
    tenant_id: str
    workspace_ids: tuple[str, ...]
    roles: tuple[str, ...]  # admin / user / auditor
    api_key_id: str | None = None  # M2M 调用时填
    trace_id: str = "-"

    def has_role(self, role: str) -> bool:
        return role in self.roles

    def in_workspace(self, workspace_id: str) -> bool:
        return workspace_id in self.workspace_ids

    def has_any_role(self, roles: Iterable[str]) -> bool:
        return any(r in self.roles for r in roles)
```

## 3.3 `libs/auth` — JWT + API Key

### 3.3.1 双轨鉴权

```python
# libs/auth/src/qzdap_auth/jwt_guard.py
from starlette.middleware.base import BaseHTTPMiddleware
from .api_key import verify_api_key
from .jwt import decode_token


class AuthMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        # 优先 API Key（M2M）
        api_key = request.headers.get("x-api-key")
        if api_key:
            principal = verify_api_key(api_key)
            principal_var.set(principal)
            return await call_next(request)

        # 其次 JWT
        auth = request.headers.get("authorization")
        if auth and auth.startswith("Bearer "):
            token = auth[7:]
            principal = decode_token(token)
            if settings.ban_mock_token and token.startswith("mock-"):
                raise IdentityMockForbidden()
            principal_var.set(principal)
            return await call_next(request)

        raise Unauthorized()
```

## 3.5 `libs/persistence` — SQLAlchemy 封装

### 3.5.1 Base / SessionFactory / UnitOfWork

```python
# libs/persistence/src/qzdap_persistence/base.py
from sqlalchemy.orm import DeclarativeBase


class Base(DeclarativeBase):
    pass


# libs/persistence/src/qzdap_persistence/session_factory.py
class SessionFactory:
    def __init__(self, url: str, *, pool_size=20, max_overflow=10, echo=False):
        self.engine = create_async_engine(
            url, pool_size=pool_size, max_overflow=max_overflow, echo=echo
        )
        self.sessionmaker = async_sessionmaker(self.engine, expire_on_commit=False)

    def session(self):
        return self.sessionmaker()

    async def close(self):
        await self.engine.dispose()


# libs/persistence/src/qzdap_persistence/unit_of_work.py
class UnitOfWork:
    def __init__(self, factory: SessionFactory):
        self._factory = factory

    async def __aenter__(self):
        self.session = self._factory.session()
        return self

    async def __aexit__(self, exc_type, exc, tb):
        try:
            if exc_type is None:
                await self.session.commit()
            else:
                await self.session.rollback()
        finally:
            await self.session.close()
```

### 3.5.2 多租户查询守卫

```python
# libs/persistence/src/qzdap_persistence/tenant_query.py
from sqlalchemy import event
from .context import current_tenant


@event.listens_for(Session, "do_orm_execute")
def _filter_by_tenant(state):
    if not state.is_select:
        return
    tenant = current_tenant()
    if tenant is None:
        return
    # 自动注入 tenant_id 过滤（约定 ORM 模型有 tenant_id 列）
    for desc in state.statement.column_descriptions:
        entity = desc.get("entity")
        if entity and hasattr(entity, "tenant_id"):
            state.statement = state.statement.where(entity.tenant_id == tenant.tenant_id)
```

> 这是"约定优于配置"：ORM 模型必须有 `tenant_id` 列。
> 不满足约定的查询会被 lint 规则警告。

### 3.6.2 进程内实现

```python
# libs/messaging/src/qzdap_messaging/in_process.py
import asyncio


class InProcessBus:
    def __init__(self, queue_size: int = 1000):
        self._queue: asyncio.Queue = asyncio.Queue(maxsize=queue_size)
        self._subs: dict[Type, list[Callable]] = {}
        self._tasks: list[asyncio.Task] = []

    async def start(self):
        self._tasks.append(asyncio.create_task(self._dispatch_loop()))

    async def stop(self):
        for t in self._tasks:
            t.cancel()
        await asyncio.gather(*self._tasks, return_exceptions=True)

    async def publish(self, event):
        await self._queue.put(event)

    def subscribe(self, event_type, handler):
        self._subs.setdefault(event_type, []).append(handler)

    async def _dispatch_loop(self):
        while True:
            event = await self._queue.get()
            for handler in self._subs.get(type(event), []):
                try:
                    await handler(event)
                except Exception:
                    log.exception("event.handler.failed", event=type(event).__name__)
            self._queue.task_done()
```

### 3.9.3 模型路由

```python
# libs/llm/src/qzdap_llm/router.py
class LLMRouter:
    """按 agent 配置选择 provider；支持 failover"""

    def __init__(self, registry: ModelRegistry, default: str):
        self._registry = registry
        self._default = default

    async def chat(self, req: LLMRequest, *, preferred: str | None = None) -> LLMResponse:
        model_id = preferred or self._default
        client = self._registry.get(model_id)
        try:
            return await client.chat(req)
        except LLMRateLimited:
            backup = self._registry.get(model_id, role="backup")
            if backup:
                return await backup.chat(req)
            raise
```

|---|---|---|
| kernel | Principal 单层 | + TenantContext + 多级 contextvars |
