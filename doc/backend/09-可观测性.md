```python
# libs/observability/src/qzdap_observability/logging.py
import structlog


def setup_logging(env: str, level: str = "INFO") -> None:
    timestamper = structlog.processors.TimeStamper(fmt="iso", utc=True)
    shared = [
        structlog.contextvars.merge_contextvars,
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        timestamper,
    ]
    if env in ("development", "demo"):
        processors = shared + [structlog.dev.ConsoleRenderer(colors=True)]
    else:
        processors = shared + [
            structlog.processors.format_exc_info,
            structlog.processors.JSONRenderer(),
        ]
    structlog.configure(
        processors=processors,
        wrapper_class=structlog.make_filtering_bound_logger(
            getattr(logging, level.upper(), logging.INFO)
        ),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(file=sys.stdout),
        cache_logger_on_first_use=True,
    )


log = structlog.get_logger()
```

## 9.3 trace_id 与 contextvars

```python
# libs/observability/src/qzdap_observability/context.py
trace_id_var: contextvars.ContextVar[str] = contextvars.ContextVar("trace_id", default="-")
request_id_var: contextvars.ContextVar[str] = contextvars.ContextVar("request_id", default="-")

def new_trace_id() -> str:
    return uuid4().hex[:16]
```

```python
# libs/http/src/qzdap_http/middleware.py
class TraceIdMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        trace_id = request.headers.get("x-trace-id") or new_trace_id()
        trace_id_var.set(trace_id)
        request_id_var.set(new_trace_id())
        response = await call_next(request)
        response.headers["X-Trace-Id"] = trace_id
        return response
```

## 9.4 OpenTelemetry 分布式追踪

### 9.4.1 初始化

```python
# libs/observability/src/qzdap_observability/tracing.py
from opentelemetry import trace
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter


def setup_tracing(
    service_name: str, otlp_endpoint: str | None = None, sample_rate: float = 1.0
) -> trace.Tracer:
    resource = Resource.create({SERVICE_NAME: service_name})
    provider = TracerProvider(resource=resource, sampler=TraceIdRatioBased(sample_rate))
    if otlp_endpoint:
        provider.add_span_processor(BatchSpanProcessor(OTLPSpanExporter(endpoint=otlp_endpoint)))
    else:
        provider.add_span_processor(BatchSpanProcessor(ConsoleSpanExporter()))
    trace.set_tracer_provider(provider)
    return trace.get_tracer(service_name)
```

### 9.5.2 HTTP 中间件埋点

```python
class MetricsMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        method = request.method
        path = request.scope.get("route").path if request.scope.get("route") else request.url.path
        tenant = current_tenant().tenant_id if current_tenant() else "-"
        start = time.monotonic()
        try:
            response = await call_next(request)
            status = response.status_code
        except Exception:
            status = 500
            raise
        finally:
            duration = time.monotonic() - start
            HTTP_REQUESTS_TOTAL.labels(
                method=method, path=path, status=status, tenant_id=tenant
            ).inc()
            HTTP_REQUEST_DURATION.labels(method=method, path=path, tenant_id=tenant).observe(
                duration
            )
        return response
```

## 9.6 健康检查

```python
# libs/http/src/qzdap_http/healthz_router.py
@router.get("/livez", include_in_schema=False)
async def livez():
    return {"status": "live"}


@router.get("/healthz", include_in_schema=False)
async def healthz(container):
    checks = {}
    overall = True

    # PG
    try:
        sf = container.resolve(SessionFactory)
        async with sf.session() as s:
            await s.execute(text("SELECT 1"))
        checks["postgres"] = "ok"
    except Exception as e:
        checks["postgres"] = f"down: {e}"
        overall = False

    # Redis
    try:
        redis = container.resolve(Redis)
        await redis.ping()
        checks["redis"] = "ok"
    except Exception as e:
        checks["redis"] = f"down: {e}"
        overall = False

    # Vector Store
    try:
        vs = container.resolve(VectorStore)
        # ping
        checks["vector_store"] = "ok"
    except Exception as e:
        checks["vector_store"] = f"down: {e}"
        overall = False

    # LLM（仅 ping 配置的 default）
    try:
        llm = container.resolve(LLMRouter)
        # ping 默认模型（用一个简单 prompt）
        checks["llm"] = "ok"
    except Exception as e:
        checks["llm"] = f"down: {e}"
        # LLM 不可用不阻塞启动；标 warning
        checks["llm"] = f"warning: {e}"

    if not overall:
        return Response(
            content=json.dumps({"status": "down", "checks": checks}),
            status_code=503,
            media_type="application/json",
        )
    return {"status": "ok", "checks": checks}


@router.get("/readyz", include_in_schema=False)
async def readyz(container):
    ensure_done = container.state.get("ensure_done", False)
    if not ensure_done:
        return Response(status_code=503, content=json.dumps({"status": "starting"}))
    return await healthz(container)
```

| 路径 | 状态码 | 含义 |
|---|---|---|
| `GET /livez` | 200 | 进程存活 |
| `GET /healthz` | 200 / 503 | 依赖全部可达 / 任一不可达 |
| `GET /readyz` | 200 / 503 | ensure 完成 + 可服务 |

## 9.8 成本计量

### 9.8.1 LLM 成本

```python
# libs/observability/src/qzdap_observability/cost.py
MODEL_PRICING = {
    "gpt-4o": {"input": 2.5 / 1_000_000, "output": 10.0 / 1_000_000},  # USD / token
    "gpt-4o-mini": {"input": 0.15 / 1_000_000, "output": 0.6 / 1_000_000},
    "claude-sonnet-5": {"input": 3.0 / 1_000_000, "output": 15.0 / 1_000_000},
}


def compute_cost(model: str, input_tokens: int, output_tokens: int) -> float:
    p = MODEL_PRICING.get(model, {"input": 0, "output": 0})
    return input_tokens * p["input"] + output_tokens * p["output"]
```

```python
# modules/agent_runtime/adapter/http/routers.py
@router.post("/sessions/{sid}/turn/stream")
async def stream_turn(sid, body, ctx=Depends(require_ctx), svc=Depends(get_agent_runtime_service)):
    SSE_ACTIVE_STREAMS.labels(path="/v1/sessions/{sid}/turn/stream", tenant_id=ctx.tenant_id).inc()

        gen = svc.run_turn_stream(ctx=ctx, session_id=sid, command=body.to_command())
```

## 9.10 关键 Grafana 面板（建议）

| 面板 | 关键指标 |
|---|---|

## 9.11 告警规则（建议）

```yaml
groups:
  - name: qzdap-app
    rules:
      - alert: HighErrorRate
        expr: |
          sum by (tenant_id) (rate(qzdap_http_requests_total{status=~"5.."}[5m]))
          /
          sum by (tenant_id) (rate(qzdap_http_requests_total[5m])) > 0.05
        for: 5m
        labels: { severity: critical }

      - alert: AgentTurnP95High
        expr: |
          histogram_quantile(0.95,
            sum by (agent_id, le) (rate(qzdap_agent_turn_duration_seconds_bucket[5m]))
          ) > 30
        for: 10m
        labels: { severity: warning }

      - alert: TenantCostSpike
        expr: |
          rate(qzdap_cost_usd_total[1h]) > 100
        for: 30m
        labels: { severity: warning }

      - alert: SSEActiveStreamsHigh
        expr: sum(qzdap_sse_active_streams) > 5000
        for: 5m
        labels: { severity: warning }

      - alert: EventDLQGrowing
        expr: redis_stream_length{stream="qzdap:events:dlq"} > 100
        for: 5m
        labels: { severity: warning }

      - alert: CrossTenantQueryBlocked
        expr: rate(qzdap_cross_tenant_query_blocked_total[5m]) > 0
```

## 9.13 反模式

| 反例 | 修正 |
|---|---|
| `print(...)` 代替日志 | 用 `log.info(...)` |
| 日志中带 token / API Key / 邮箱 | structlog processor 过滤 |
| trace_id 仅日志中含，跨服务不可关联 | `x-trace-id` header + OTel |
| 指标无 `tenant_id` label | 所有业务指标强制带 |
| `/metrics` 需要鉴权 | 永远不要鉴权 |
| 健康检查查所有外部依赖 | 仅查关键路径（PG / Redis / Vector） |
| OTel exporter 配成 ConsoleExporter 在生产 | 用 OTLP gRPC → Collector |
| 成本计量散落在各模块 | 集中在 `observability_module` |

## 9.14 与原模板的关键差异

| 维度 | 原模板 | Enterprise-Agent-OS |
|---|---|---|
| 指标 label | status / path / method | + `tenant_id`（强制） |
| 业务指标 | workflow / LLM tokens | + turn / tool / skill / memory / cost |
| 成本 | 无 | LLM / Tool / Skill 用量与成本计量 |
| 多租户 | 无 | 跨租户查询被阻止时埋点告警 |
| LLM 用量 | 仅 token 计数 | + 按模型定价 USD 成本 |
