# 03 · 共享内核（packages/* · FSD shared 层 · Hex SDK）

> 6 个 shared packages：`@qzdap/web-ui` `@qzdap/web-api` `@qzdap/web-mock` `@qzdap/web-types` `@qzdap/web-hooks` `@qzdap/web-utils`
> FSD shared 层定位；Hex SDK 定位（api 是 HttpApiAdapter 底层 SDK；mock 是 MockApiAdapter 底层 SDK）
> 业务代码通过 features/<bc>/api/adapters/ 间接使用；**不直接 import mock**
> DTO 类型与后端 OpenAPI 一一对应（TypeScript Codegen）

> **基于 docs/web-refactor/03-共享包.md 模板**。
> 与原模板差异：包名 `@qzdap/*`（区别于 `@de/*`）；DTO 来自后端 OpenAPI Codegen；mock 镜像后端真实 handler。

### 3.3.2 ApiClient（精简 · 无 mock）

```ts
// packages/api/src/client.ts
export interface ApiClientConfig {
  baseUrl: string;
  getToken: () => string | null;
  getTenantId: () => string | null;
  getWorkspaceId: () => string | null;
}

export class ApiClient {
  constructor(private readonly config: ApiClientConfig) {}

  async request<T>(opts: {
    method?: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';
    path: string;
    body?: unknown;
    query?: Record<string, string>;
    headers?: Record<string, string>;
    signal?: AbortSignal;
  }): Promise<T> {
    const url = new URL(this.config.baseUrl + opts.path);
    if (opts.query) Object.entries(opts.query).forEach(([k, v]) => url.searchParams.set(k, v));

    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      ...opts.headers,
    };
    const token = this.config.getToken();
    if (token) headers['Authorization'] = `Bearer ${token}`;
    const tenantId = this.config.getTenantId();
    if (tenantId) headers['X-Tenant-Id'] = tenantId;
    const workspaceId = this.config.getWorkspaceId();
    if (workspaceId) headers['X-Workspace-Id'] = workspaceId;

    const traceId = getCurrentTraceId();
    if (traceId) headers['X-Trace-Id'] = traceId;

    const res = await fetch(url.toString(), {
      method: opts.method ?? 'GET',
      headers,
      body: opts.body ? JSON.stringify(opts.body) : undefined,
      signal: opts.signal,
    });
    if (!res.ok) {
      const body = await res.json().catch(() => ({}));
      throw mapStatusToError(res.status, body);
    }
    return this.parseResponse<T>(res);
  }

  /** SSE 流式 */
  async *stream(opts: {
    method: 'POST';
    path: string;
    body?: unknown;
    signal?: AbortSignal;
  }): AsyncIterable<unknown> {
    const res = await this.fetchStream(opts);
    const reader = res.body!.getReader();
    const decoder = new TextDecoder();
    let buffer = '';
    while (true) {
      const { value, done } = await reader.read();
      if (done) break;
      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n\n');
      buffer = lines.pop() ?? '';
      for (const line of lines) {
        if (line.startsWith('data:')) {
          const payload = line.slice(5).trim();
          if (payload && payload !== '[DONE]') yield JSON.parse(payload);
        }
      }
    }
  }

  private async fetchStream(opts: { method: string; path: string; body?: unknown; signal?: AbortSignal }): Promise<Response> {
    const headers = this.buildHeaders(opts);
    return fetch(this.config.baseUrl + opts.path, {
      method: opts.method,
      headers: { ...headers, Accept: 'text/event-stream' },
      body: opts.body ? JSON.stringify(opts.body) : undefined,
      signal: opts.signal,
    });
  }
}
```

> **关键**：ApiClient **纯粹是 HttpApiAdapter 底层 SDK**；**没有 mock 调用**。

### 3.3.3 业务域客户端（与后端 modules/ 一一对应）

```ts
// packages/api/src/domains/session.ts
import type { SessionDTO, TurnChunkDTO } from '@qzdap/web-types';
import type { ApiClient } from '../client';

export function createSessionApi(client: ApiClient) {
  return {
    listSessions: (workspaceId: string) =>
      client.request<SessionDTO[]>({ path: `/v1/sessions`, query: { workspaceId } }),

    getSession: (id: string) =>
      client.request<SessionDTO>({ path: `/v1/sessions/${id}` }),

    createSession: (input: { agentVersionId: string; workspaceId: string }) =>
      client.request<SessionDTO>({ method: 'POST', path: `/v1/sessions`, body: input }),

    closeSession: (id: string) =>
      client.request<SessionDTO>({ method: 'DELETE', path: `/v1/sessions/${id}` }),

    streamTurn: function* (id: string, input: { content: string; traceId: string }) {
      return client.stream({
        method: 'POST',
        path: `/v1/sessions/${id}/turns`,
        body: input,
      }) as AsyncIterable<TurnChunkDTO>;
    },
  };
}

export type SessionApi = ReturnType<typeof createSessionApi>;
```

> **对应后端 modules/session/* RESTful + SSE 路由**（详见后端 `06-接口规范`）。

```ts
import type { SessionDTO, TurnChunkDTO } from '@qzdap/web-types';
import { sessionFixtures } from '../fixtures/session';

export const sessionHandlers = {
  listSessions: async (workspaceId: string): Promise<SessionDTO[]> =>
    sessionFixtures.sessions.filter(s => s.workspaceId === workspaceId),

  getSession: async (id: string): Promise<SessionDTO | null> =>
    sessionFixtures.sessions.find(s => s.id === id) ?? null,

  createSession: async (input: { agentVersionId: string; workspaceId: string }): Promise<SessionDTO> => ({
    id: `ses_${Math.random().toString(36).slice(2, 10)}`,
    agentVersionId: input.agentVersionId,
    workspaceId: input.workspaceId,
    status: 'idle',
    createdAt: new Date().toISOString(),
    turns: [],
  }),

  closeSession: async (id: string): Promise<SessionDTO> => {
    const s = sessionFixtures.sessions.find(s => s.id === id);
    if (!s) throw new Error(`Session ${id} not found`);
    return { ...s, status: 'closed' };
  },

  streamTurn: async function* (id: string, input: { content: string }): AsyncIterable<TurnChunkDTO> {
    yield { type: 'message', delta: '正在思考...' };
    await delay(300);
    yield { type: 'tool_call', toolName: 'search', args: { q: input.content } };
    await delay(500);
    yield { type: 'message', delta: '已找到结果：' };
    yield { type: 'message', delta: input.content.split('').reverse().join('') };
    yield { type: 'done', usage: { input: 10, output: 20 } };
  },
};

function delay(ms: number) { return new Promise(r => setTimeout(r, ms)); }
```

### 3.11.2 Entity 在 BC model 层

```ts
// features/session/model/entities/Session.ts
import type { SessionDTO } from '@qzdap/web-types';
import { SessionId } from '../value-objects/SessionId';
import { Turn } from './Turn';

export class Session {
  private _status: SessionStatus = 'idle';
  private readonly _turns: Turn[] = [];

  constructor(private readonly _id: SessionId) {}

  get id(): SessionId { return this._id; }
  get status(): SessionStatus { return this._status; }
  get turns(): readonly Turn[] { return this._turns; }

  appendTurn(turn: Turn): void {
    this._status = 'running';
    this._turns.push(turn);
  }

  close(): void {
    this.assertOpen();
    this._status = 'closed';
  }

  private assertOpen(): void {
    if (this._status === 'closed') throw new Error(`Session ${this._id.value} is closed`);
  }

  /** 适配器层调用 */
  static fromDTO(dto: SessionDTO): Session {
    const s = new Session(new SessionId(dto.id));
    if (dto.status === 'closed') s.close();
    dto.turns.forEach(t => s.appendTurn(Turn.fromDTO(t)));
    return s;
  }

  toDTO(): SessionDTO {
    return {
      id: this._id.value,
      agent_version_id: '',
      workspace_id: '',
      status: this._status,
      created_at: '',
      turns: this._turns.map(t => t.toDTO()),
    };
  }
}
```

> **DTO → Entity 在 adapter 边界**；Entity 含业务规则；DTO 是传输格式。
