> 8 个 Bounded Context 与后端 modules/ 1:1 对齐（agent_runtime / session / skill / tool / knowledge / memory / governance / identity）+ channel 前端独有；

### 4.4.1 model 层

```ts
// features/session/model/entities/Session.ts
import type { SessionDTO, TurnChunkDTO } from '@qzdap/web-types';
import { SessionId } from '../value-objects/SessionId';
import { SessionStatus } from '../value-objects/SessionStatus';
import { Turn } from './Turn';

export class Session {
  private _status: SessionStatus = 'idle';
  private readonly _turns: Turn[] = [];

  constructor(
    private readonly _id: SessionId,
    private readonly _agentVersionId: string,
    private readonly _workspaceId: string,
  ) {}

  get id(): SessionId { return this._id; }
  get agentVersionId(): string { return this._agentVersionId; }
  get workspaceId(): string { return this._workspaceId; }
  get status(): SessionStatus { return this._status; }
  get turns(): readonly Turn[] { return this._turns; }

  /** 业务方法：append turn（含不变量） */
  appendTurn(turn: Turn): void {
    this.assertOpen();
    this._status = 'running';
    this._turns.push(turn);
  }

  /** 业务方法：append chunk 到当前 turn */
  appendChunk(turnId: number, chunk: TurnChunkDTO): void {
    this.assertOpen();
    const turn = this._turns.find(t => t.id === turnId);
    if (!turn) throw new Error(`Turn ${turnId} not found`);
    turn.append(chunk);
  }

  close(): void {
    this.assertOpen();
    this._status = 'closed';
  }

  private assertOpen(): void {
    if (this._status === 'closed') throw new Error(`Session ${this._id.value} is closed`);
  }

  static fromDTO(dto: SessionDTO): Session {
    const s = new Session(
      new SessionId(dto.id),
      dto.agent_version_id,
      dto.workspace_id,
    );
    if (dto.status === 'closed') s.close();
    dto.turns.forEach(t => s.appendTurn(Turn.fromDTO(t)));
    return s;
  }

  toDTO(): SessionDTO {
    return {
      id: this._id.value,
      agent_version_id: this._agentVersionId,
      workspace_id: this._workspaceId,
      status: this._status,
      created_at: '',
      turns: this._turns.map(t => t.toDTO()),
    };
  }
}
```

### 4.4.2 lib 层（Use Case）

```ts
// features/session/lib/use-cases/SendTurnUseCase.ts
import type { SessionRepository, SessionStreamPort } from '../../model/repositories/SessionRepository';
import type { SessionEventBus } from '../../model/events/SessionEventBus';
import type { SessionId } from '../../model/value-objects/SessionId';

export interface SendTurnDeps {
  sessionRepo: SessionRepository;
  streamPort: SessionStreamPort;
  eventBus: SessionEventBus;
}

export class SendTurnUseCase {
  constructor(private readonly deps: SendTurnDeps) {}

  /** 流式执行 turn */
  async *execute(sessionId: SessionId, content: string): AsyncIterable<unknown> {
    await this.deps.eventBus.publish({
      type: 'session.turn.started',
      occurredAt: new Date().toISOString(),
      sessionId,
      turnId: Date.now(),
    });

    let turnId = 0;
    let usage: { input: number; output: number } | null = null;
    for await (const chunk of this.deps.streamPort.stream(sessionId, {
      content,
      traceId: crypto.randomUUID(),
    })) {
      yield chunk;
      if (typeof chunk === 'object' && chunk !== null) {
        const c = chunk as { type?: string; usage?: { input: number; output: number } };
        if (c.type === 'message' && c.usage) usage = c.usage;
        if (c.type === 'done' && c.usage) usage = c.usage;
      }
      turnId++;
    }

    await this.deps.eventBus.publish({
      type: 'session.turn.completed',
      occurredAt: new Date().toISOString(),
      sessionId,
      turnId,
      usage: usage ?? { input: 0, output: 0 },
    });
  }
}
```

```ts
// features/session/lib/hooks/useSendTurn.ts
import { useCallback } from 'react';
import { useMutation } from '@tanstack/react-query';
import { useDeps } from '@/app/composition/useDeps';
import { SendTurnUseCase } from '../use-cases/SendTurnUseCase';
import { SessionId } from '../../model/value-objects/SessionId';

export function useSendTurn() {
  const { session } = useDeps();
  const useCase = new SendTurnUseCase({
    sessionRepo: session.sessionRepo,
    streamPort: session.streamPort,
    eventBus: session.eventBus,
  });

  return useMutation({
    mutationFn: async (vars: { sessionId: string; content: string }) => {
      const chunks: unknown[] = [];
      for await (const chunk of useCase.execute(new SessionId(vars.sessionId), vars.content)) {
        chunks.push(chunk);
      }
      return chunks;
    },
  });
}
```

```ts
// features/session/api/adapters/HttpApiSessionAdapter.ts（生产）
import type { SessionRepository, SessionStreamPort } from '../../model/repositories/SessionRepository';
import type { Session } from '../../model/entities/Session';
import type { SessionId } from '../../model/value-objects/SessionId';
import { Session as SessionEntity } from '../../model/entities/Session';
import { api } from '@qzdap/web-api';

export class HttpApiSessionAdapter implements SessionRepository, SessionStreamPort {
  async findById(id: SessionId): Promise<Session | null> {
    try {
      const dto = await api.session.getSession(id.value);
      return SessionEntity.fromDTO(dto);
    } catch (e) {
      if ((e as { status?: number }).status === 404) return null;
      throw e;
    }
  }

  async list(workspaceId: string): Promise<readonly Session[]> {
    const dtos = await api.session.listSessions(workspaceId);
    return dtos.map(SessionEntity.fromDTO);
  }

  async save(session: Session): Promise<void> {
    await api.session.updateSession(session.id.value, session.toDTO());
  }

  async create(input: { agentVersionId: string; workspaceId: string }): Promise<Session> {
    const dto = await api.session.createSession(input);
    return SessionEntity.fromDTO(dto);
  }

  async *stream(id: SessionId, input: { content: string; traceId: string }): AsyncIterable<unknown> {
    yield* api.session.streamTurn(id.value, input);
  }
}
```

```ts
      workspace_id: input.workspaceId,
      status: 'idle',
      created_at: new Date().toISOString(),
      turns: [],
    };
    storage.set(`session:${dto.id}`, dto);
    return SessionEntity.fromDTO(dto);
  }
```

```tsx
      chunks={chunks}
```

```ts
// features/skill/model/entities/SkillInvocation.ts
export class SkillInvocation {
  constructor(
    private readonly _id: SkillInvocationId,
    private readonly _skillPackageId: SkillPackageId,
    private _status: 'pending' | 'running' | 'completed' | 'failed',
    private _result?: unknown,
    private _error?: string,
    private _startedAt?: string,
    private _finishedAt?: string,
  ) {}

  get id(): SkillInvocationId { return this._id; }
  get status(): 'pending' | 'running' | 'completed' | 'failed' { return this._status; }
  get result(): unknown { return this._result; }
  get error(): string | undefined { return this._error; }

  /** 业务方法：开始 */
  start(): void {
    if (this._status !== 'pending') throw new Error(`Cannot start: status=${this._status}`);
    this._status = 'running';
    this._startedAt = new Date().toISOString();
  }

  /** 业务方法：完成 */
  complete(result: unknown): void {
    if (this._status !== 'running') throw new Error(`Cannot complete: status=${this._status}`);
    this._status = 'completed';
    this._result = result;
    this._finishedAt = new Date().toISOString();
  }

  /** 业务方法：失败 */
  fail(error: string): void {
    if (this._status !== 'running') throw new Error(`Cannot fail: status=${this._status}`);
    this._status = 'failed';
    this._error = error;
    this._finishedAt = new Date().toISOString();
  }
}
```

关键 Use Case：`CheckPolicyUseCase`（拦截所有调用）、`CreateApprovalUseCase`、`SignApprovalUseCase`、`ListAuditLogsUseCase`

**特殊**：governance 是**横切 BC**——其他 BC 的 Use Case 必须先过 governance 拦截（在 lib/use-cases/ 编排中）。

## 4.14 跨 BC 通信三选项

### 4.14.1 读其他 BC 数据（经本 BC Repository Port）

```ts
// 例：session BC 需要查 agent_runtime 的 AgentVersion
// 在 session/lib/use-cases/ 编排中注入 AgentVersionRepository Port
import type { AgentVersionRepository } from '@/features/agent_runtime/model/repositories/AgentVersionRepository';

export interface CreateSessionDeps {
  agentVersionRepo: AgentVersionRepository;  // ← 跨 BC 依赖（经 Port）
  sessionRepo: SessionRepository;
}

export class CreateSessionUseCase {
  constructor(private readonly deps: CreateSessionDeps) {}

  async execute(input: { agentVersionId: string; workspaceId: string }): Promise<Session> {
    const av = await this.deps.agentVersionRepo.findById(new AgentVersionId(input.agentVersionId));
    if (!av) throw new Error('AgentVersion not found');
    return this.deps.sessionRepo.create({ agentVersionId: av.id.value, workspaceId: input.workspaceId });
  }
}
```

> **关键**：session 通过 Port 调 agent_runtime —— **Use Case 不感知 agent_runtime 的具体实现**。

## 4.15 ACL 翻译规则

每个 BC 内 `model/acl/<other-bc>.ts`：

```ts
// features/session/model/acl/agent_runtime.ts
import type { AgentVersionSummaryDTO } from '@qzdap/web-types';

export function mapAgentVersionToSessionInput(
  av: AgentVersionSummaryDTO,
  workspaceId: string,
): { agentVersionId: string; workspaceId: string } {
  if (av.status !== 'published') {
    throw new Error(`AgentVersion ${av.id} is not published (status=${av.status})`);
  }
  return { agentVersionId: av.id, workspaceId };
}
```

```ts
// features/governance/model/acl/session.ts
import type { SessionSummaryDTO } from '@qzdap/web-types';

export function mapSessionToAuditEntry(
  session: SessionSummaryDTO,
  action: string,
): AuditEntryInput {
  return {
    actor: 'system',
    target: { type: 'session', id: session.id },
    action,
    metadata: { workspaceId: session.workspace_id },
  };
}
```

> **ACL 函数在 model 层**；**翻译规则也是业务规则**——必须有测试。

```ts
}
```
