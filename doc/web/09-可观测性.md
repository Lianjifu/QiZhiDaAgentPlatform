# 09 · 可观测性（前端 trace · metric · log · cost · quality · 上报）

> 前端可观测性横切所有 BC；与后端 `modules/observability/` 平行；
> trace_id 贯穿 SSE Turn；前端 reporter 上报 metric / log / event；
> widgets/observability 提供 TraceViewer / CostDashboard / QualityDashboard；
> 与后端 qizhida-agent-platform/observability/ 同步；前端只发不存。

> **基于 docs/web-refactor/09-测试与CI.md 模板**。
> 与原模板差异：聚焦前端可观测性；trace_id 贯穿 SSE；reporter 上报到后端 observability 模块；与后端 `docs/qizhida-agent-platform/09-可观测性.md` 平行阅读。

## 9.2 trace_id 管理

### 9.2.1 trace 上下文

```ts
// web/src/app/observability/trace.ts
import { traceContext } from '@/app/observability/trace';

export class TraceContext {
  private current: string | null = null;
  private spans: TraceSpan[] = [];

  start(traceId?: string): string {
    this.current = traceId ?? crypto.randomUUID();
    sessionStorage.setItem('eos:trace_id', this.current);
    return this.current;
  }

  end(): void {
    if (this.current) {
      observabilityReporter.reportTrace({
        trace_id: this.current,
        spans: this.spans,
        ended_at: new Date().toISOString(),
      });
      this.spans = [];
    }
    this.current = null;
    sessionStorage.removeItem('eos:trace_id');
  }

  get(): string | null {
    return this.current;
  }

  span(name: string, attributes?: Record<string, unknown>): SpanScope {
    const span: TraceSpan = {
      id: crypto.randomUUID(),
      trace_id: this.current!,
      name,
      attributes,
      started_at: new Date().toISOString(),
    };
    this.spans.push(span);
    return {
      span,
      end: (extra?: Record<string, unknown>) => {
        span.finished_at = new Date().toISOString();
        if (extra) Object.assign(span.attributes ??= {}, extra);
      },
    };
  }
}

export interface TraceSpan {
  id: string;
  trace_id: string;
  name: string;
  attributes?: Record<string, unknown>;
  started_at: string;
  finished_at?: string;
}

export interface SpanScope {
  span: TraceSpan;
  end: (extra?: Record<string, unknown>) => void;
}

export const traceContext = new TraceContext();
```

// features/session/lib/use-cases/SendTurnUseCase.ts
import { traceContext } from '@/app/observability/trace';

export class SendTurnUseCase {
  async *execute(sessionId: SessionId, content: string): AsyncIterable<unknown> {
    const traceId = traceContext.start();
    const span = traceContext.span('session.send_turn', {
      session_id: sessionId.value,
      content_length: content.length,
    });

    try {
      for await (const chunk of this.deps.streamPort.stream(sessionId, {
        content,
        traceId,
      })) {
        yield chunk;
      }
      span.end({ status: 'completed' });
    } catch (e) {
      span.end({ status: 'failed', error: String(e) });
      throw e;
    } finally {
      traceContext.end();
    }
  }
}

### 9.3.2 metric reporter

```ts
// web/src/app/observability/metric.ts
import { observabilityReporter } from './reporter';

export class MetricReporter {
  private buffer: MetricDTO[] = [];
  private flushTimer: number | null = null;

  record(name: MetricName, value: number, tags: Record<string, string> = {}): void {
    this.buffer.push({
      name,
      value,
      tags,
      timestamp: new Date().toISOString(),
    });
    if (this.buffer.length >= 50) this.flush();
    else if (this.flushTimer === null) {
      this.flushTimer = window.setTimeout(() => this.flush(), 5000);
    }
  }

  flush(): void {
    if (this.buffer.length === 0) return;
    observabilityReporter.reportBatch(this.buffer.splice(0));
    if (this.flushTimer !== null) {
      window.clearTimeout(this.flushTimer);
      this.flushTimer = null;
    }
  }
}

export const metricReporter = new MetricReporter();
window.addEventListener('beforeunload', () => metricReporter.flush());
```

### 9.3.3 自动埋点

```ts
// features/session/lib/hooks/useSendTurn.ts（续）
import { metricReporter } from '@/app/observability/metric';

return useMutation({
  mutationFn: async (vars: { sessionId: string; content: string }) => {
    const start = performance.now();
    const chunks: unknown[] = [];
    let usage: { input: number; output: number } | null = null;
    let toolInvocations = 0;
    let skillInvocations = 0;

    for await (const chunk of useCase.execute(new SessionId(vars.sessionId), vars.content)) {
      chunks.push(chunk);
      const c = chunk as { type?: string };
      if (c.type === 'tool_call') toolInvocations++;
      if (c.type === 'skill_invocation') skillInvocations++;
      if (c.type === 'done' && 'usage' in c) usage = (c as any).usage;
    }

    const duration = performance.now() - start;
    metricReporter.record('turn.duration_ms', duration, { session_id: vars.sessionId });
    if (usage) {
      metricReporter.record('turn.tokens.input', usage.input, { session_id: vars.sessionId });
      metricReporter.record('turn.tokens.output', usage.output, { session_id: vars.sessionId });
      metricReporter.record('turn.cost_usd', estimateCost(usage), { session_id: vars.sessionId });
    }
    metricReporter.record('turn.tool_invocations', toolInvocations, { session_id: vars.sessionId });
    metricReporter.record('turn.skill_invocations', skillInvocations, { session_id: vars.sessionId });

    return chunks;
  },
});

function estimateCost(usage: { input: number; output: number }): number {
  // 按模型单价估算（远期接入后端真实价格）
  const inputRate = 0.01 / 1000;
  const outputRate = 0.03 / 1000;
  return usage.input * inputRate + usage.output * outputRate;
}
```

### 9.7.1 视图组件

```tsx
// web/src/widgets/observability/TraceViewer.tsx
import { useQuery } from '@tanstack/react-query';
import { api } from '@qzdap/web-api';
import type { TraceDTO, TraceSpanDTO } from '@qzdap/web-types';

export function TraceViewer({ traceId }: { traceId: string }) {
  const { data: trace, isLoading } = useQuery({
    queryKey: ['trace', traceId],
    queryFn: () => api.observability.getTrace(traceId),
  });

  if (isLoading) return <Skeleton />;
  if (!trace) return <EmptyState message="Trace not found" />;

  return (
    <div className="space-y-2">
      <div className="text-sm text-gray-500">
        trace_id: <code>{trace.trace_id}</code> · duration: {trace.duration_ms}ms
      </div>
      <Timeline>
        {trace.spans.map((span) => (
          <SpanRow key={span.id} span={span} />
        ))}
      </Timeline>
    </div>
  );
}

function SpanRow({ span }: { span: TraceSpanDTO }) {
  const duration = span.finished_at
    ? new Date(span.finished_at).getTime() - new Date(span.started_at).getTime()
    : 0;
  return (
    <div className="flex items-center gap-2 border-l-2 border-blue-500 pl-3">
      <span className="font-mono">{span.name}</span>
      <span className="text-xs text-gray-500">{duration}ms</span>
      {span.attributes && (
        <details className="text-xs">
          <summary>attributes</summary>
          <pre>{JSON.stringify(span.attributes, null, 2)}</pre>
        </details>
      )}
    </div>
  );
}
```

## 9.11 用户隐私与脱敏

| 字段 | 处理 |
|---|---|
| 用户输入文本 | 上报 trace 时**摘要**（hash + 前 100 字符） |
| 用户邮箱 / 姓名 | log 中**不记录** |
| token | log 中**不记录**（自动过滤） |
| 密码 / 密钥 | 自动 redact |

```ts
// web/src/app/observability/sanitizer.ts
const SENSITIVE_KEYS = ['password', 'token', 'secret', 'authorization', 'cookie'];

export function sanitize<T>(data: T): T {
  if (typeof data !== 'object' || data === null) return data;
  if (Array.isArray(data)) return data.map(sanitize) as unknown as T;
  const result: any = {};
  for (const [key, value] of Object.entries(data)) {
    if (SENSITIVE_KEYS.some(k => key.toLowerCase().includes(k))) {
      result[key] = '[REDACTED]';
    } else {
      result[key] = sanitize(value);
    }
  }
  return result;
}
```
