## 8.2 `@qzdap/web-utils/storage`

### 8.2.1 统一入口

```ts
// packages/utils/src/storage.ts
import { z } from 'zod';

const STORAGE_KEY_PREFIX = 'eos:';

export class Storage<T> {
  constructor(
    private readonly key: string,
    private readonly schema: z.ZodType<T>,
    private readonly options: { encrypt?: boolean; ttlMs?: number } = {},
  ) {}

  get(): T | null {
    try {
      const raw = localStorage.getItem(STORAGE_KEY_PREFIX + this.key);
      if (!raw) return null;
      const parsed = JSON.parse(raw);
      if (this.options.ttlMs && Date.now() - parsed.__t > this.options.ttlMs) {
        this.delete();
        return null;
      }
      const validated = this.schema.safeParse(parsed.v);
      return validated.success ? validated.data : null;
    } catch {
      return null;
    }
  }

  set(value: T): void {
    const payload = {
      __t: Date.now(),
      v: value,
    };
    localStorage.setItem(STORAGE_KEY_PREFIX + this.key, JSON.stringify(payload));
  }

  delete(): void {
    localStorage.removeItem(STORAGE_KEY_PREFIX + this.key);
  }

  list<T>(prefix: string): readonly T[] {
    const fullPrefix = STORAGE_KEY_PREFIX + prefix;
    const items: T[] = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key?.startsWith(fullPrefix)) {
        const raw = localStorage.getItem(key);
        if (raw) {
          try {
            const parsed = JSON.parse(raw);
            items.push(parsed.v);
          } catch {}
        }
      }
    }
    return items;
  }

  clear(prefix: string): void {
    const fullPrefix = STORAGE_KEY_PREFIX + prefix;
    const keys: string[] = [];
    for (let i = 0; i < localStorage.length; i++) {
      const key = localStorage.key(i);
      if (key?.startsWith(fullPrefix)) keys.push(key);
    }
    keys.forEach(k => localStorage.removeItem(k));
  }
}

export const storage = {
  get: <T>(key: string, schema?: z.ZodType<T>): T | null =>
    schema ? new Storage(key, schema).get() : (JSON.parse(localStorage.getItem(STORAGE_KEY_PREFIX + key) ?? 'null')),
  set: <T>(key: string, value: T): void => new Storage(key, z.any()).set(value),
  delete: (key: string): void => new Storage(key, z.any()).delete(),
  list: <T>(prefix: string): readonly T[] => new Storage<unknown>('', z.any()).list(prefix),
};
```

### 8.3.1 auth store

```ts
// web/src/entities/auth/store.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import { TokenSchema, UserSchema } from '@qzdap/web-utils/storage';
import type { UserDTO } from '@qzdap/web-types';

export interface AuthState {
  token: string | null;
  user: UserDTO | null;
  expiresAt: number | null;

  setToken: (token: string, expiresAt: number) => void;
  setUser: (user: UserDTO) => void;
  refresh: (token: string, expiresAt: number) => void;
  logout: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      token: null,
      user: null,
      expiresAt: null,

      setToken: (token, expiresAt) => set({ token, expiresAt }),
      setUser: (user) => set({ user }),
      refresh: (token, expiresAt) => set({ token, expiresAt }),
      logout: () => set({ token: null, user: null, expiresAt: null }),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({
        token: state.token,
        user: state.user,
        expiresAt: state.expiresAt,
      }),
      onRehydrateStorage: () => (state) => {
        // 校验 token 过期
        if (state?.expiresAt && Date.now() > state.expiresAt) {
          state.logout();
        }
      },
    },
  ),
);
```

## 8.5 离线模式（LocalStorage Adapter · Hex 第三类）

### 8.5.1 离线检测

```ts
// web/src/hooks/useOnline.ts
import { useEffect, useState } from 'react';

export function useOnline(): boolean {
  const [online, setOnline] = useState(navigator.onLine);
  useEffect(() => {
    const handleOnline = () => setOnline(true);
    const handleOffline = () => setOnline(false);
    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);
    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);
  return online;
}

export function useOfflineMode(): boolean {
  return !useOnline();
}
```

### 8.5.2 Composition Root 模式决策

```ts
// web/src/app/composition/resolveMode.ts
import { useOfflineMode } from '@/hooks/useOnline';
import { useAuthStore } from '@/entities/auth/store';
import { isMockToken } from '@/dev/mock-tokens';

export function resolveMode(): AdapterMode {
  // 1. 离线 → LocalStorage
  if (!navigator.onLine) return 'local';

  return 'http';
}
```

### 8.5.3 LocalStorage Adapter（Hex 第三类）

```ts
// features/session/api/adapters/LocalStorageSessionAdapter.ts
import type { SessionRepository, SessionStreamPort } from '../../model/repositories/SessionRepository';
import type { Session } from '../../model/entities/Session';
import type { SessionId } from '../../model/value-objects/SessionId';
import { Session as SessionEntity } from '../../model/entities/Session';
import { storage } from '@qzdap/web-utils/storage';
import type { SessionDTO, TurnChunkDTO } from '@qzdap/web-types';

export class LocalStorageSessionAdapter implements SessionRepository {
  async findById(id: SessionId): Promise<Session | null> {
    const dto = storage.get<SessionDTO>(`session:${id.value}`);
    return dto ? SessionEntity.fromDTO(dto) : null;
  }

  async list(_workspaceId: string): Promise<readonly Session[]> {
    return storage.list<SessionDTO>('session:').map(SessionEntity.fromDTO);
  }

  async save(session: Session): Promise<void> {
    storage.set(`session:${session.id.value}`, session.toDTO());
  }

  async create(input: { agentVersionId: string; workspaceId: string }): Promise<Session> {
    const dto: SessionDTO = {
      id: `ses_${Math.random().toString(36).slice(2, 10)}`,
      agent_version_id: input.agentVersionId,
      workspace_id: input.workspaceId,
      status: 'idle',
      created_at: new Date().toISOString(),
      closed_at: null,
      turns: [],
      metadata: {},
    };
    storage.set(`session:${dto.id}`, dto);
    return SessionEntity.fromDTO(dto);
  }
}
```

> **关键**：LocalStorageSessionAdapter 实现 `SessionRepository` Port——Use Case 完全不感知用了哪个。

### 8.5.4 离线场景

| 场景 | 行为 |
|---|---|
| 断网时访问 Session 列表 | LocalStorageSessionAdapter.list() 返回本地缓存 |
| 断网时创建 Session | LocalStorageSessionAdapter.create() 写入本地 |
| 联网后 | sync queue（远期 P5）把本地写同步到后端 |
| 弱网 | 双调用，本地优先 + 后台同步 |

## 8.6 数据迁移（schema 版本化）

### 8.6.1 版本化存储

```ts
// web/src/entities/auth/migration.ts
const MIGRATIONS = {
  '1.0.0': (data: any) => data,
  '1.1.0': (data: any) => ({
    ...data,
    expiresAt: data.expiresAt ?? Date.now() + 3600 * 1000,
  }),
  '2.0.0': (data: any) => {
    const { tenantId, ...rest } = data;
    return { ...rest, tenantId: tenantId ?? localStorage.getItem('eos:lastTenantId') };
  },
};

const CURRENT_VERSION = '2.0.0';

export function migrateAuth(data: unknown): AuthState | null {
  if (!data || typeof data !== 'object') return null;
  const stored = data as { __version?: string; state?: any };
  let state = stored.state;
  if (!state) return null;
  const versions = Object.keys(MIGRATIONS).sort();
  const startIdx = versions.indexOf(stored.__version ?? '1.0.0');
  for (let i = startIdx; i < versions.length; i++) {
    state = MIGRATIONS[versions[i] as keyof typeof MIGRATIONS](state);
  }
  return state as AuthState;
}
```

### 8.7.2 mock-bridge 安装

```ts
// web/src/dev/mock-bridge.ts
import { dispatchMockRequest } from '@qzdap/web-mock';
import { api } from '@qzdap/web-api';
import { isMockToken } from './mock-tokens';

export function installMockBridge(getToken: () => string | null): void {
  if (import.meta.env.MODE === 'production') return;  // 生产不装

  const USE_MOCK = import.meta.env.VITE_USE_MOCK === '1';

  // 拦截 ApiClient.request
  const client = api.client as any;
  const originalRequest = client.request?.bind(client);
  if (!originalRequest) return;

  client.request = async function (opts: { method?: string; path: string; body?: unknown }) {
    const token = getToken();
    const wantMock = USE_MOCK || (token && isMockToken(token));
    if (wantMock) {
      return dispatchMockRequest(opts.method ?? 'GET', opts.path, opts.body);
    }
    return originalRequest(opts);
  };

  // 拦截 ApiClient.stream
  if (client.stream) {
    const originalStream = client.stream.bind(client);
    client.stream = async function* (opts: { method: 'POST'; path: string; body?: unknown }) {
      const token = getToken();
      const wantMock = USE_MOCK || (token && isMockToken(token));
      if (wantMock) {
        yield* mockStreamTurn(opts.path, opts.body);
      } else {
        yield* originalStream(opts);
      }
    };
  }
}

async function* mockStreamTurn(path: string, body: any): AsyncIterable<unknown> {
  // 解析 path: /v1/sessions/{id}/turns
  const match = path.match(/\/v1\/sessions\/([^/]+)\/turns/);
  if (!match) throw new Error(`Unknown stream path: ${path}`);
  const sessionId = match[1];
  yield* mock.session.streamTurn(sessionId, body);
}
```

## 8.11 离线写同步（远期 P5）

```ts
// web/src/app/infrastructure/sync-queue.ts（远期）
interface PendingWrite<T> {
  endpoint: string;
  method: 'POST' | 'PUT' | 'DELETE';
  body: T;
  queuedAt: number;
}

export class SyncQueue {
  private queue: PendingWrite<unknown>[] = [];

  enqueue(write: PendingWrite<unknown>): void {
    this.queue.push(write);
    localStorage.setItem('sync-queue', JSON.stringify(this.queue));
  }

  async flush(): Promise<void> {
    if (!navigator.onLine) return;
    while (this.queue.length > 0) {
      const write = this.queue[0];
      try {
        await api.client.request({
          method: write.method,
          path: write.endpoint,
          body: write.body,
        });
        this.queue.shift();
      } catch (e) {
        break;
      }
    }
    localStorage.setItem('sync-queue', JSON.stringify(this.queue));
  }
}
```
